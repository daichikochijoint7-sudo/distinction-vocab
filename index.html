スプレッドシート & GASの作成手順・コードを見る
  1. Googleスプレッドシートを新規作成し、1行目のヘッダーを以下に設定します：
    A: Word | B: Meaning | C: Example | D: Note | E: Status | F: UpdatedAt
  2. 「拡張機能」>「Apps Script」を開き、以下のコードを貼り付けて保存します。
  3. 「デプロイ」>「新しいデプロイ」> 種類の選択で「ウェブアプリ」を選択。
  4. アクセスできるユーザーを「全員（Anyone）」にしてデプロイし、生成されたURLを上に貼り付けます。
const SHEET_NAME = 'words';

// 単語一覧の取得 (GET)
function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    // 'words' シートがない場合は1番目のシートを自動参照（クラッシュ防止）
    const sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
    const rows = sheet.getDataRange().getValues();
    const data = [];

    for (let i = 1; i < rows.length; i++) {
      const row = rows[i];
      if (!row[0]) continue; // 単語が空の行は除外
      data.push({
        rowIndex: i + 1,
        word: String(row[0] || '').trim(),
        meaning: String(row[1] || '').trim(),
        example: String(row[2] || '').trim(),
        note: String(row[3] || '').trim(),
        status: String(row[4] || '高頻度').trim()
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: data }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ステータス更新の書き込み (POST)
function doPost(e) {
  try {
    const params = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
    const rowIndex = Number(params.rowIndex);
    const newStatus = params.status;

    // E列 (ステータス) と F列 (更新日時) を更新
    sheet.getRange(rowIndex, 5).setValue(newStatus);
    sheet.getRange(rowIndex, 6).setValue(new Date());

    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
「Failed to fetch」が出る場合のチェックポイント
  • アクセス権限：GASの「デプロイを管理」で「アクセスできるユーザー」が「全員（Anyone）」になっているか確認（「自分のみ」だとCORSエラーになります）。
  • デプロイの更新：コード修正後は「新しいデプロイ」または「デプロイを管理」→「編集」→「新バージョン」を選択して保存してください。
  • シート名：スプレッドシートのタブ名が「words」になっているか確認（上記修正コードを貼ればタブ名が何でも自動認識します）。
